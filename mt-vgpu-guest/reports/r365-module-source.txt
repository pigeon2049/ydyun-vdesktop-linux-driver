/* r365: single TA marker probe (DM=3/4, opcode 0x66) via raw firmware-queue
 * submit. Session validation mirrors mt_live_marker.c. The marker store
 * bookkeeping (pending/count) is deliberately untouched, so any firmware
 * completion event lands harmlessly in the trial event buffer.
 * Minimal stimulus: one 0x50-byte packet, opcode only, no workload payload,
 * no descriptor, no resource mapping. Observes accept (COMPLETE event) vs
 * reject (FAULT / no-response) signatures, then unloads immediately.
 */
#include <linux/sched.h>
#include "../../../kernel/mt_guest_device.h"

static bool enable;
module_param(enable, bool, 0400);
static u32 target_dm = 3;
module_param(target_dm, uint, 0400);
MODULE_PARM_DESC(target_dm, "Target DM (3=TA candidate, 4=fallback)");
static u32 opcode = 0x66;
module_param(opcode, uint, 0400);
MODULE_PARM_DESC(opcode, "Firmware opcode (0x66=RGXVertex/UniversalQueue candidate)");

/* Observables, read-only from sysfs after insmod returns. */
static int result = -ENODATA;
module_param(result, int, 0444);
static u32 wire_used;
module_param(wire_used, uint, 0444);
static bool consumed;
module_param(consumed, bool, 0444);
static u32 ev_w0, ev_w1, ev_w2, ev_w3, ev_w4, ev_w5;
module_param(ev_w0, uint, 0444);
module_param(ev_w1, uint, 0444);
module_param(ev_w2, uint, 0444);
module_param(ev_w3, uint, 0444);
module_param(ev_w4, uint, 0444);
module_param(ev_w5, uint, 0444);

/* Same layout guard as mt_live_marker.c: rejects any other probe build. */
static_assert(sizeof(struct mt_guest_device) == 30784);
static_assert(offsetof(struct mt_guest_device, service) == 29552);
static_assert(offsetof(struct mt_guest_device, runtime) == 29712);
static_assert(offsetof(struct mt_guest_device, markers) == 29888);

#define TA_PROBE_WIRE_ID 0x65036501u
#define TA_PROBE_POLL_MS 10
#define TA_PROBE_POLL_TRIES 200 /* 2s total, bounded */

static void ta_peek_event(struct mt_fw_queue_io *q, u32 dm, u32 idx, u32 *w)
{
	u32 base = dm * MT_FW_DM_BYTES + MT_FW_EVENT_OFFSET +
		   idx * MT_FW_EVENT_BYTES;
	int i;

	for (i = 0; i < 6; i++)
		w[i] = mt_fw_io_read32(q, base + (u32)(i * 4));
}

static int __init mt_live_ta_marker_init(void)
{
	struct pci_dev *pdev = NULL;
	struct module *owner = NULL;
	struct mt_guest *g;
	struct mt_guest_device *d;
	struct mt_marker_store *s;
	struct mt_fw_queue_io *q;
	u8 packet[MT_FW_COMMAND_BYTES];
	u32 dm, i;
	u32 cmd_cur, ev_cur, cmd_tail0, ev_tail0;
	u32 w[6] = {0, 0, 0, 0, 0, 0};
	bool found = false;
	bool ev_err = false;
	int ret = -ENODEV;

	if (!enable)
		return -EPERM;
	if (target_dm < 1 || target_dm >= MT_FW_DM_COUNT)
		return -EOPNOTSUPP;
	dm = target_dm;

	pdev = mt_guest_find_s3000();
	if (!pdev)
		return -ENODEV;
	device_lock(&pdev->dev);
	if (!mt_guest_match_s3000(pdev->vendor, pdev->device,
				 pdev->subsystem_vendor,
				 pdev->subsystem_device) ||
	    !pdev->driver || strcmp(pdev->driver->name, MT_GUEST_DRIVER_NAME))
		goto unlock_device;
	owner = pdev->driver->driver.owner;
	if (!owner || !try_module_get(owner)) {
		owner = NULL;
		goto unlock_device;
	}
	g = pci_get_drvdata(pdev);
	if (!g)
		goto unlock_device;
	d = container_of(g, struct mt_guest_device, state);
	s = &d->markers;
	mutex_lock(&g->trial_lock);
	ret = -EBUSY;
	if (!d->runtime.published || d->runtime.event_result ||
	    !g->trial.pinned || !g->trial.connected || !d->service.running ||
	    s->lock != &g->trial_lock || s->queue != &g->firmware_queue ||
	    s->opaque != g || !s->ops || !s->can_submit ||
	    s->total || s->ready || s->work_ready)
		goto unlock_session;
	for (i = 0; i < MT_FW_DM_COUNT; i++)
		if (s->count[i] || !list_empty(&s->pending[i]))
			goto unlock_session;
	ret = s->can_submit(g);
	if (ret)
		goto unlock_session;
	q = s->queue;

	cmd_cur = dm * MT_FW_DM_BYTES + MT_FW_CURSOR_OFFSET;
	ev_cur = dm * MT_FW_DM_BYTES + MT_FW_CURSOR_OFFSET + 32;
	cmd_tail0 = mt_fw_io_read32(q, cmd_cur + 8);
	ev_tail0 = mt_fw_io_read32(q, ev_cur + 8);
	if (cmd_tail0 >= MT_FW_RING_ENTRIES || ev_tail0 >= MT_FW_RING_ENTRIES) {
		ret = -EIO;
		goto unlock_session;
	}

	/* Minimal marker: opcode only. No command_va, no root_pa, no workload. */
	memset(packet, 0, sizeof(packet));
	mt_fw_put32(packet, 0x0c, opcode);
	mt_fw_put32(packet, 0x48, TA_PROBE_WIRE_ID);
	mt_fw_put32(packet, 0x4c, (u32)current->pid);
	wire_used = TA_PROBE_WIRE_ID;

	ret = mt_fw_queue_try_submit(q, dm, 0, packet);
	/* try_submit errors happen before all hardware writes. */
	if (ret)
		goto unlock_session;
	mutex_unlock(&g->trial_lock);
	device_unlock(&pdev->dev);

	/* Bounded observation window; locks released, MMIO reads only.
	 * Events are peeked, never acknowledged: the driver's own poll
	 * drains them normally afterwards.
	 */
	for (i = 0; i < TA_PROBE_POLL_TRIES; i++) {
		u32 ev_tail = mt_fw_io_read32(q, ev_cur + 8);
		u32 idx;

		if (ev_tail >= MT_FW_RING_ENTRIES) {
			ev_err = true;
			break;
		}
		for (idx = ev_tail0; idx != ev_tail;
		     idx = (idx + 1) & (MT_FW_RING_ENTRIES - 1)) {
			ta_peek_event(q, dm, idx, w);
			if (w[2] == TA_PROBE_WIRE_ID) {
				found = true;
				break;
			}
		}
		if (found)
			break;
		msleep(TA_PROBE_POLL_MS);
	}
	/* Consumption check is informational only. */
	consumed = (mt_fw_io_read32(q, cmd_cur + 8) != cmd_tail0);
	if (ev_err) {
		ret = -EIO;
	} else if (found) {
		ev_w0 = w[0]; ev_w1 = w[1]; ev_w2 = w[2];
		ev_w3 = w[3]; ev_w4 = w[4]; ev_w5 = w[5];
		if (w[1] == 0)
			ret = 0;	/* COMPLETE: firmware accepted */
		else if (w[1] == 0x101)
			ret = -EIO;	/* FAULT: firmware rejected */
		else
			ret = -EPROTO;	/* unexpected event kind */
	} else {
		ret = -ETIMEDOUT;	/* no completion event observed */
	}
	goto done;

unlock_session:
	mutex_unlock(&g->trial_lock);
unlock_device:
	device_unlock(&pdev->dev);
done:
	result = ret;
	pr_info("mt_live_ta_marker: dm=%u opcode=0x%x wire=0x%x result=%d consumed=%d ev=[%08x %08x %08x %08x %08x %08x]\n",
		dm, opcode, wire_used, result, (int)consumed,
		ev_w0, ev_w1, ev_w2, ev_w3, ev_w4, ev_w5);
	if (owner)
		module_put(owner);
	if (pdev)
		pci_dev_put(pdev);
	return 0;
}

static void __exit mt_live_ta_marker_exit(void) {}
module_init(mt_live_ta_marker_init);
module_exit(mt_live_ta_marker_exit);
MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("r365: single TA marker probe (DM/opcode) via raw fw-queue submit");
